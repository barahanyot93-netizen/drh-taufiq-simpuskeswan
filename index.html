// Fitur Lupa Kata Sandi document.getElementById('forgot-password-link').addEventListener('click', function(e) { e.preventDefault(); const emailField = document.getElementById('login-email') || document.getElementById('email'); const emailValue = emailField ? emailField.value.trim() : ''; if (!emailValue) { alert("Silakan isi alamat email Anda di kolom Email terlebih dahulu."); if (emailField) emailField.focus(); return; } firebase.auth().sendPasswordResetEmail(emailValue) .then(function() { alert("Email reset kata sandi telah dikirim ke: " + emailValue + "\n\nSilakan periksa folder Inbox atau Spam pada email Anda."); }) .catch(function(error) { let pesan = "Gagal mengirim email reset."; if (error.code === 'auth/user-not-found') { pesan = "Email tidak terdaftar di sistem."; } else if (error.code === 'auth/invalid-email') { pesan = "Format email tidak valid."; } alert(pesan); }); });
